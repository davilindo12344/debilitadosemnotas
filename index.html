⚠️ Tabela perfis não encontrada

Execute o SQL abaixo no Supabase → SQL Editor para criar a tabela e ativar perfis e ranking:

-- 1. Cria tabela de perfis
CREATE TABLE IF NOT EXISTS public.perfis (
  user_id  uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  nome     text,
  apelido  text,
  foto     text,
  serie    text,
  media    numeric,
  pontos   integer DEFAULT 0,
  updated_at timestamptz DEFAULT now()
);

-- 2. Habilita RLS
ALTER TABLE public.perfis ENABLE ROW LEVEL SECURITY;

-- 3. Política: cada usuário gerencia o próprio perfil
CREATE POLICY "users_own_perfil" ON public.perfis
  FOR ALL USING (auth.uid() = user_id)
  WITH CHECK (auth.uid() = user_id);

-- 4. Política: todos os autenticados podem ler (para o ranking)
CREATE POLICY "read_all_perfis" ON public.perfis
  FOR SELECT USING (auth.role() = 'authenticated');
